s = input("Enter a string: ")

and_result = ""
xor_result = ""
print("\nCharacter|ASCII|AND|XOR")

for c in s:
    ascii_value = ord(c)   
    
    and_val = ascii_value & 127
    and_char = chr(and_val)
    and_result += and_char
    
    xor_val = ascii_value ^ 127
    xor_char = chr(xor_val)
    xor_result += xor_char
    print(f"{c:^7} | {ascii_value:^4} | {and_val:^3} | {xor_val:^3}")

print("After AND with 127 :", and_result)
print("After XOR with 127 :", repr(xor_result))
